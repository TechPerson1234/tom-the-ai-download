    <h3>Tom The AI Lite (All Devices Supported)</h3>
    <div id="chat">
        <div class='msg ai'>Tom: System online. Maximum backward compatibility engaged.</div>
    </div>
    <input type="text" id="in" placeholder="Ask Tom...">
    <button id="btn" onclick="ask()">Send</button>

    <script type="text/javascript">
        var chat = document.getElementById('chat');
        var input = document.getElementById('in');
        var btn = document.getElementById('btn');

        var WORKER_URL = "https://tom-the-ai-api.wen471942.workers.dev/";

        function ask() {
            var val = input.value;

            val = val ? val.replace(/^\s+|\s+$/g, '') : '';
            if(!val) return;

            chat.innerHTML += "<div class='msg user'>Me: " + val + "</div>";
            input.value = "";
            btn.disabled = true;

            if (chat.scrollHeight) { chat.scrollTop = chat.scrollHeight; }

            var xhr;
            if (window.XMLHttpRequest) {
                xhr = new XMLHttpRequest();
            } else {
                try {
                    xhr = new ActiveXObject("Msxml2.XMLHTTP");
                } catch(e) {
                    try { xhr = new ActiveXObject("Microsoft.XMLHTTP"); } catch(e2) {
                        chat.innerHTML += "<div class='msg error'>Error: This device does not support web requests.</div>";
                        btn.disabled = false;
                        return;
                    }
                }
            }

            var encoder = window.encodeURIComponent ? encodeURIComponent : escape;
            xhr.open("GET", WORKER_URL + "?prompt=" + encoder(val), true);

            xhr.onreadystatechange = function() {
                if (xhr.readyState === 4) {
                    btn.disabled = false;

                    if (xhr.status === 200 || xhr.status === 0) {
                        try {
                            var data;

                            if (window.JSON && window.JSON.parse) {
                                data = JSON.parse(xhr.responseText);
                            } else {
                                data = eval('(' + xhr.responseText + ')');
                            }

                            if (data && data.status === "ok") {
                                chat.innerHTML += "<div class='msg ai'>Tom: " + data.response + "</div>";
                            } else {
                                var msg = data && data.message ? data.message : "Unknown API Error";
                                chat.innerHTML += "<div class='msg error'>Error: " + msg + "</div>";
                            }
                        } catch (err) {
                            chat.innerHTML += "<div class='msg error'>Data Error: Unable to read response.</div>";
                        }
                    } else {
                        chat.innerHTML += "<div class='msg error'>Connection Failed: Status " + xhr.status + ". This is likely a TLS/SSL security block on this device.</div>";
                    }

                    if (chat.scrollHeight) { chat.scrollTop = chat.scrollHeight; }
                }
            };

            xhr.onerror = function() {
                btn.disabled = false;
                chat.innerHTML += "<div class='msg error'>Network Error: Connection refused by device security.</div>";
            };

            xhr.send(null);
        }

        input.onkeypress = function(e) {
            var charCode = (e && e.which) ? e.which : (window.event ? window.event.keyCode : 0);
            if(charCode == 13) { ask(); return false; }
        };
    </script>
